# Demo GUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A local web page, served by the demo server at `GET /`, that shows the feed for a chosen seed reader with weights and quotas set on the page.

**Architecture:** A new module `src/http/demo.ts` registers three routes on the existing Fastify app. `POST /demo/feed` lays the posted feed weights and quotas over the default config, validates it with `parseConfig`, builds an engine for that request and returns the engine's response plus a small `articles` map the page needs. The page is one static HTML file with plain JavaScript. The engine does not change.

**Tech Stack:** unchanged: Node.js 22.23 running `.ts` directly, TypeScript ~6.0.3, npm, Vitest 5, Zod 4, Fastify 5, ESLint 10 with typescript-eslint 8. The page uses no library and no build step.

**Spec:** `docs/superpowers/specs/2026-10-06-demo-gui-design.md`. Ruling D31 in
`DECISIONS.md` (with D17). This plan runs on branch `how-it-works-diagram`, which
already holds the spec and D31.

## Global Constraints

- Package manager: npm (D1). Verify command: `npm run verify` (type check, ESLint, Vitest).
- Node runs `.ts` directly: relative imports end in `.ts`; types are imported with `import type` (or inline `type`); no `enum`, `namespace` or constructor parameter properties.
- TypeScript `strict`. No `any` in any source file.
- The engine (`src/engine`) does not change in this plan. Engine code never reads the clock; the demo layer passes `now`.
- One business rule lives in one place (AGENTS.md section 5): weight and quota value rules stay in `parseConfig` (`src/engine/config.ts`); the demo route checks only the body's outer shape.
- The page renders everything received from the server with `textContent` or created elements, never `innerHTML`, and loads nothing from outside the demo server: no fonts, scripts or styles from other hosts (D31).
- Demo routes are local only: the server keeps listening on 127.0.0.1 (D17, D31).
- Writing: plain language, no emoji, never an en dash or em dash. Comments explain a non-obvious decision only.
- Public repository: no client-internal material, real data or personal data.

```ts
import { readFileSync } from "node:fs";
import type { FastifyInstance } from "fastify";
import { z } from "zod";
import type { Repository } from "../data/repository.ts";
import { parseConfig, type Config } from "../engine/config.ts";
import { compareIds } from "../engine/order.ts";
import { RequestError, createRecommender } from "../engine/recommend.ts";
import type { Article, ArticleId, UserContext } from "../engine/types.ts";

export interface DemoDeps {
  repository: Repository;
  /** config/default.json as parsed JSON, before validation. */
  rawConfig: unknown;
  /** The seed readers the page can choose from. */
  users: UserContext[];
  topics: string[];
  now: () => Date;
}

const DEMO_LIMIT = 20;
const HOUR_MS = 3_600_000;

export function topicsOf(articles: Article[]): string[] {
  return [...new Set(articles.flatMap((a) => a.topics))].sort(compareIds);
}

/** Local-only demo routes (D31): the page, its options, and a feed built from posted weights and quotas. */
export function registerDemo(app: FastifyInstance, deps: DemoDeps): void {
  const page = readFileSync(new URL("./demo.html", import.meta.url), "utf8");
  const base = parseConfig(deps.rawConfig);
  // Only the outer shape is checked here; weight and quota values are parseConfig's job.
  const feedBody = z.strictObject({
    reader: z.number().int().min(0).max(deps.users.length - 1),
    seed: z.unknown().optional(),
    weights: z.unknown(),
    quotas: z.unknown(),
  });

  app.get("/", async (_request, reply) => reply.type("text/html; charset=utf-8").send(page));

  app.get("/demo/options", async () => ({
    readers: deps.users.map((user) => ({ label: user.userId ?? "Anonymous (cold start)", user })),
    topics: deps.topics,
    defaults: { weights: base.profiles.feed.weights, quotas: base.profiles.feed.quotas },
  }));

  app.post("/demo/feed", async (request, reply) => {
    const body = feedBody.safeParse(request.body);
    if (!body.success) return reply.code(400).send({ error: z.prettifyError(body.error) });
    const { reader, seed, weights, quotas } = body.data;

    let config: Config;
    try {
      config = parseConfig({ ...base, profiles: { ...base.profiles, feed: { ...base.profiles.feed, weights, quotas } } });
    } catch (error) {
      return reply.code(400).send({ error: error instanceof Error ? error.message : String(error) });
    }

    const now = deps.now();
    try {
      const response = await createRecommender({ repository: deps.repository, config }).recommend({
        mode: "feed",
        user: deps.users[reader],
        limit: DEMO_LIMIT,
        now,
        seed,
      });
      const found = await deps.repository.getArticles(response.items.map((i) => i.articleId));
      const articles: Record<ArticleId, { publisherId: string; topics: string[]; ageHours: number }> = {};
      for (const a of found) {
        articles[a.id] = {
          publisherId: a.publisherId,
          topics: a.topics,
          ageHours: Math.round((now.getTime() - a.publishedAt.getTime()) / HOUR_MS),
        };
      }
      return { ...response, articles };
    } catch (error) {
      if (error instanceof RequestError) return reply.code(400).send({ error: error.message });
      throw error;
    }
  });
}
```

<script>
  const CRITERIA = [
    ["recency", "Freshness"],
    ["popularity", "Popularity"],
    ["topics", "Interest match"],
    ["follows", "Follows"],
  ];
  const SHARES = [
    ["publishers", "Followed publishers"],
    ["communities", "Followed communities"],
    ["surprise", "Surprise"],
  ];
  const MAX_SEED = 4294967295;
  const byId = (id) => document.getElementById(id);
  const topicInputs = new Map();
  let options;
  let timer;
  let latest = 0;

  /** Builds an element; strings become text nodes, so nothing is parsed as HTML. */
  function el(tag, attrs, ...children) {
    const node = document.createElement(tag);
    for (const [key, value] of Object.entries(attrs || {})) {
      if (key === "class") node.className = value;
      else node.setAttribute(key, value);
    }
    node.append(...children);
    return node;
  }

  function slider(id, label, max, step) {
    const input = el("input", { type: "range", id, min: "0", max: String(max), step: String(step) });
    const out = el("output", { for: id, id: id + "-value" });
    return el("div", { class: "slider" }, el("label", { for: id }, label), out, input);
  }

  function buildControls() {
    const reader = byId("reader");
    options.readers.forEach((r, i) => reader.append(el("option", { value: String(i) }, r.label)));
    for (const [key, label] of CRITERIA) byId("weights").append(slider("w-" + key, label, 2, 0.1));
    for (const [key, label] of SHARES) byId("shares").append(slider("s-" + key, label, 1, 0.05));
    options.topics.forEach((topic, i) => {
      const input = el("input", { type: "number", id: "topic-" + i, min: "0", max: "1", step: "0.05", value: "0" });
      topicInputs.set(topic, input);
      byId("topics").append(el("div", { class: "topic" }, el("label", { for: "topic-" + i }, topic), input));
    });
    for (const [key, label] of CRITERIA) {
      byId("bar-legend").append(el("span", { class: "key" }, el("i", { style: "background:var(--c-" + key + ")" }), label));
    }
  }

  function applyDefaults() {
    const { weights, quotas } = options.defaults;
    for (const [key] of CRITERIA) byId("w-" + key).value = String(weights[key] ?? 0);
    const q = quotas || { enabled: false, topics: {}, follows: { publishers: 0, communities: 0 }, surprise: 0 };
    byId("quotas-on").checked = q.enabled;
    byId("s-publishers").value = String(q.follows.publishers);
    byId("s-communities").value = String(q.follows.communities);
    byId("s-surprise").value = String(q.surprise);
    for (const [topic, input] of topicInputs) input.value = String(q.topics[topic] ?? 0);
  }

  function readSettings() {
    const weights = Object.fromEntries(CRITERIA.map(([key]) => [key, Number(byId("w-" + key).value)]));
    const topics = {};
    for (const [topic, input] of topicInputs) {
      const share = Number(input.value);
      if (share > 0) topics[topic] = share;
    }
    const body = {
      reader: Number(byId("reader").value),
      weights,
      quotas: {
        enabled: byId("quotas-on").checked,
        topics,
        follows: { publishers: Number(byId("s-publishers").value), communities: Number(byId("s-communities").value) },
        surprise: Number(byId("s-surprise").value),
      },
    };
    const seedText = byId("seed").value.trim();
    if (seedText !== "") body.seed = Number(seedText);
    return body;
  }

  function updateLabels() {
    for (const input of document.querySelectorAll('input[type="range"]')) {
      byId(input.id + "-value").textContent = Number(input.value).toFixed(2);
    }
    const { quotas } = readSettings();
    const total = Object.values(quotas.topics).reduce((sum, s) => sum + s, 0)
      + quotas.follows.publishers + quotas.follows.communities + quotas.surprise;
    const totalEl = byId("total");
    totalEl.textContent = "Total of all shares: " + total.toFixed(2) + " of 1";
    totalEl.classList.toggle("over", total > 1 + 1e-9);
    byId("quota-fields").setAttribute("aria-disabled", String(!quotas.enabled));

    const r = options.readers[Number(byId("reader").value)].user;
    const interests = Object.entries(r.topicInterests).map(([t, v]) => t + " " + v).join(", ") || "none";
    const follows = [...r.followedPublishers, ...r.followedCommunities].join(", ") || "none";
    byId("reader-summary").textContent =
      "Reads " + [r.language, ...r.fallbackLanguages].join(", ") + ". Interests: " + interests + ". Follows: " + follows + ".";
  }

  function age(hours) {
    return hours < 48 ? hours + " h old" : Math.round(hours / 24) + " days old";
  }

  function renderFeed(response) {
    const list = byId("feed");
    list.replaceChildren();
    if (response.items.length === 0) list.append(el("li", { class: "empty" }, "No articles match this reader."));
    for (const item of response.items) {
      const a = response.articles[item.articleId];
      const details = a
        ? [item.articleId, a.publisherId, a.topics.join(", ") || "no topics", age(a.ageHours)].join(" · ")
        : item.articleId;
      const reasons = el("div", { class: "reasons" }, ...item.reasons.map((r) => el("span", { class: "reason" }, r)));
      const main = el("div", { class: "main" },
        el("div", { class: "line" }, el("span", { class: "chip " + item.slotType }, item.slotType), el("span", { class: "article" }, details)),
        reasons);
      const score = el("span", { class: "score" }, item.finalScore === null ? "pinned" : item.finalScore.toFixed(3));
      const row = el("li", { class: "item" + (item.slotType === "pin" ? " pin" : "") },
        el("span", { class: "rank" }, String(item.rank)), main, score);
      const entries = Object.entries(item.breakdown);
      if (entries.length > 0) {
        const totalWeight = entries.reduce((sum, [, e]) => sum + e.weight, 0);
        const bar = el("div", { class: "bar", role: "img", "aria-label": "Score made of: " + entries.map(([k, e]) => k + " " + e.contribution.toFixed(2)).join(", ") });
        for (const [key, e] of entries) {
          const share = totalWeight > 0 ? (e.contribution / totalWeight) * 100 : 0;
          bar.append(el("span", { style: "width:" + share.toFixed(2) + "%;background:var(--c-" + key + ")", title: key + " " + e.score.toFixed(3) + " x " + e.weight }));
        }
        row.append(bar);
      }
      list.append(row);
    }
    byId("meta").textContent = response.meta.candidateCount + " candidates, seed " + response.meta.seed;
  }

  function showError(message) {
    const box = byId("error");
    box.textContent = message;
    box.hidden = message === "";
  }

  async function refresh() {
    const id = ++latest;
    byId("feed").classList.add("stale");
    let res;
    let body;
    try {
      res = await fetch("/demo/feed", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify(readSettings()),
      });
      body = await res.json();
    } catch {
      if (id === latest) {
        showError("The demo server did not answer. Is npm start still running?");
        byId("feed").classList.remove("stale");
      }
      return;
    }
    if (id !== latest) return;
    byId("feed").classList.remove("stale");
    if (!res.ok) {
      showError(body.error || "The server answered " + res.status + ".");
      return;
    }
    showError("");
    renderFeed(body);
  }

  function schedule() {
    updateLabels();
    clearTimeout(timer);
    timer = setTimeout(refresh, 200);
  }

  function newSeed() {
    byId("seed").value = String(Math.floor(Math.random() * (MAX_SEED + 1)));
  }

  async function start() {
    try {
      options = await (await fetch("/demo/options")).json();
    } catch {
      showError("Could not load the demo options. Is npm start running?");
      return;
    }
    buildControls();
    applyDefaults();
    newSeed();
    byId("settings").addEventListener("input", schedule);
    byId("settings").addEventListener("submit", (event) => event.preventDefault());
    byId("new-seed").addEventListener("click", () => { newSeed(); schedule(); });
    byId("reset").addEventListener("click", () => { applyDefaults(); schedule(); });
    updateLabels();
    refresh();
  }

  start();
</script>
</body>
</html>
